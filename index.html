app.css
/* one declaration, BOTH
   directions: --rev is 0 or 1,
   so it collapses to 1, or to
   -1 */
.gbtn.is-go .g {
  animation-delay: calc(
    var(--step) * var(--rev)
    + var(--rev) * (var(--n)
    - 1 - 2 * var(--i)));
}

/* a round blur under a scaleX
   comes out stretched along x;
   one directional smear out of
   one blur and one scale, then
   STEPPED across while unseen */
@keyframes g-drive {
  42% { transform: translateX(
    -1.5em) scaleX(2.3);
    filter: blur(.1em);
    opacity: 0;
    animation-timing-function:
      steps(1, end); }
}
app.js
function play(btn, fromRight) {
  // restarted mid-flight, a CSS
  // animation begins again at
  // 0%: a letter at its apex
  // would teleport to the floor
  if (btn.dataset.busy) return;
  // which half of the
  // button the pointer crossed
  btn.style.setProperty('--rev',
    fromRight ? 1 : 0);
  btn.classList.add('is-go');
  // a cancelled run REJECTS its
  // finished promise, so
  // allSettled, and not all
  const runs = btn.getAnimations(
    { subtree: true }).filter(
      (a) => a.animationName);
  btn.dataset.busy = '1';
  Promise.allSettled(runs.map(
    (a) => a.finished)).then(() => {
      btn.classList.remove('is-go');
      delete btn.dataset.busy;
  });
}
